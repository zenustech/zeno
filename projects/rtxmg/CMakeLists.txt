cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ZenoClusterAssets LANGUAGES C CXX)
endif()
get_filename_component(ZENO_CLUSTER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Retire the alternate source path from existing standalone experiment builds.
unset(ZENO_CLUSTER_MESHOPT_SOURCE_DIR CACHE)
if(NOT EXISTS "${ZENO_CLUSTER_ROOT}/projects/meshoptimizer/src/meshoptimizer.h")
  message(FATAL_ERROR "Initialize the CLAS dependency: git submodule update --init projects/meshoptimizer")
endif()
set(MESHOPT_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MESHOPT_BUILD_GLTFPACK OFF CACHE BOOL "" FORCE)
set(MESHOPT_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ZENO_CLUSTER_ROOT}/projects/meshoptimizer" meshoptimizer EXCLUDE_FROM_ALL)
if(NOT EXISTS "${ZENO_CLUSTER_ROOT}/projects/zstd/lib/zstd.h")
  message(FATAL_ERROR "Initialize the cluster page codec: git submodule update --init projects/zstd")
endif()
# Build the codec without its install/uninstall targets, which conflict with
# OpenVDB's uninstall target in the parent project.
file(GLOB ZENO_CLUSTER_ZSTD_SOURCES CONFIGURE_DEPENDS
  "${ZENO_CLUSTER_ROOT}/projects/zstd/lib/common/*.c"
  "${ZENO_CLUSTER_ROOT}/projects/zstd/lib/compress/*.c"
  "${ZENO_CLUSTER_ROOT}/projects/zstd/lib/decompress/*.c")
add_library(zeno_cluster_zstd STATIC ${ZENO_CLUSTER_ZSTD_SOURCES})
target_include_directories(zeno_cluster_zstd PUBLIC "${ZENO_CLUSTER_ROOT}/projects/zstd/lib")
target_compile_definitions(zeno_cluster_zstd PRIVATE ZSTD_DISABLE_ASM ZSTD_LEGACY_SUPPORT=0)
add_library(zeno_cluster_cache STATIC ClusterCache.cpp ClusterRtxmg.cpp)
target_compile_features(zeno_cluster_cache PUBLIC cxx_std_17)
target_include_directories(zeno_cluster_cache PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}" "${ZENO_CLUSTER_ROOT}/zeno/tpls/include")
target_link_libraries(zeno_cluster_cache PUBLIC meshoptimizer zeno_cluster_zstd)
target_include_directories(zeno_cluster_cache PRIVATE "${ZENO_CLUSTER_ROOT}/projects/zstd/lib")
add_executable(zeno_cluster_bake ClusterBake.cpp)
target_link_libraries(zeno_cluster_bake PRIVATE zeno_cluster_cache)
target_include_directories(zeno_cluster_bake PRIVATE "${ZENO_CLUSTER_ROOT}/projects/meshoptimizer/demo")
add_executable(zeno_cluster_repack ClusterRepack.cpp)
add_executable(zeno_cluster_import_rtxmg ClusterImportRtxmg.cpp)
add_executable(zeno_cluster_quality ClusterQuality.cpp)
add_executable(zeno_cluster_coverage_test ClusterCoverageTest.cpp)
add_executable(zeno_cluster_adaptive_lod_test ClusterAdaptiveLodTest.cpp)
target_compile_features(zeno_cluster_adaptive_lod_test PRIVATE cxx_std_17)
target_link_libraries(zeno_cluster_coverage_test PRIVATE zeno_cluster_cache)
target_link_libraries(zeno_cluster_quality PRIVATE zeno_cluster_cache)
target_link_libraries(zeno_cluster_import_rtxmg PRIVATE zeno_cluster_cache)
target_link_libraries(zeno_cluster_repack PRIVATE zeno_cluster_cache)
target_include_directories(zeno_cluster_repack PRIVATE "${ZENO_CLUSTER_ROOT}/projects/zstd/lib")
find_package(CUDAToolkit REQUIRED)
if(NOT CMAKE_CUDA_COMPILER_LOADED)
  enable_language(CUDA)
endif()
add_library(zeno_cluster_coverage_gpu STATIC ClusterCoverageGpu.cu)
target_link_libraries(zeno_cluster_coverage_gpu PUBLIC zeno_cluster_cache CUDA::cudart_static)
set_target_properties(zeno_cluster_coverage_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
  CUDA_ARCHITECTURES "75-real;89-real;75-virtual")
target_compile_options(zeno_cluster_coverage_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--generate-line-info>)
target_link_libraries(zeno_cluster_quality PRIVATE zeno_cluster_coverage_gpu)
add_executable(zeno_cluster_coverage_gpu_test ClusterCoverageGpuTest.cpp)
target_link_libraries(zeno_cluster_coverage_gpu_test PRIVATE zeno_cluster_coverage_gpu)
add_library(zeno_cluster_runtime STATIC ClusterRuntime.cpp ClusterGpu.cu)
set_target_properties(zeno_cluster_runtime PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
  CUDA_ARCHITECTURES "75-real;89-real;75-virtual")
target_compile_options(zeno_cluster_runtime PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--generate-line-info>)
target_include_directories(zeno_cluster_runtime PUBLIC
  "${ZENO_CLUSTER_ROOT}/zenovis/xinxinoptix/include")
target_link_libraries(zeno_cluster_runtime PUBLIC zeno_cluster_cache CUDA::cudart_static CUDA::cuda_driver)
target_compile_definitions(zeno_cluster_runtime PUBLIC NOMINMAX)
add_executable(zeno_cluster_smoke ClusterSmoke.cpp)
target_link_libraries(zeno_cluster_smoke PRIVATE zeno_cluster_runtime CUDA::nvrtc)
target_compile_definitions(zeno_cluster_smoke PRIVATE
  CLUSTER_OPTIX_INCLUDE="${ZENO_CLUSTER_ROOT}/zenovis/xinxinoptix/include"
  CLUSTER_CUDA_INCLUDE="${CUDAToolkit_INCLUDE_DIRS}")
add_executable(zeno_cluster_validate ClusterValidate.cpp)
target_link_libraries(zeno_cluster_validate PRIVATE zeno_cluster_runtime CUDA::nvrtc)
target_compile_definitions(zeno_cluster_validate PRIVATE
  CLUSTER_OPTIX_INCLUDE="${ZENO_CLUSTER_ROOT}/zenovis/xinxinoptix/include"
  CLUSTER_CUDA_INCLUDE="${CUDAToolkit_INCLUDE_DIRS}")
if(MSVC)
  target_compile_options(zeno_cluster_cache PRIVATE /bigobj)
  target_compile_options(zeno_cluster_bake PRIVATE /bigobj)
  target_compile_options(zeno_cluster_quality PRIVATE /bigobj)
endif()
